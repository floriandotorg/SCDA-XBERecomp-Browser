# Native game hooks over browser memory. src/browser/hook-compat comes first on their include path and shadows the
# native kernel and d3d headers, so src/game sources build unchanged; symbols holds the image's image_symbols.h.
function(browser_game_hooks target hooks symbols)
    set(sources)
    foreach(hook IN LISTS hooks)
        if(EXISTS ${root}/src/game/${hook}.cpp)
            list(APPEND sources ${root}/src/game/${hook}.cpp)
        else()
            list(APPEND sources ${root}/src/xdk/${hook}.cpp)
        endif()
    endforeach()
    # browser stand-ins for native files the selected hooks call into, built the same way
    if("hud" IN_LIST hooks)
        list(APPEND sources ${root}/src/browser/loading-screen.cpp)
    endif()
    if(BROWSER_MENU)
        list(APPEND sources ${root}/src/browser/loading-input.cpp ${root}/src/browser/exit-menu.cpp)
    endif()
    add_library(${target} OBJECT ${sources})
    target_include_directories(${target} BEFORE PRIVATE ${root}/src/browser/hook-compat)
    target_include_directories(${target} PRIVATE ${symbols})
    target_link_libraries(${target} PRIVATE browser_flags)
    target_compile_definitions(${target} PRIVATE X86_COMPACT_MEMORY ${browser_hook_definitions})
    if(BROWSER_FLAT_MEMORY)
        target_compile_definitions(${target} PRIVATE X86_FLAT_MEMORY)
    endif()
    target_compile_options(${target} PRIVATE -O3)
    if(BROWSER_WASM_SIMD)
        target_compile_options(${target} PRIVATE -msimd128)
    endif()
    if(BROWSER_MISSION AND BROWSER_ACTUAL_TAIL_CALLS)
        target_compile_options(${target} PRIVATE -mtail-call)
    endif()
endfunction()

function(browser_publish_boot)
    set(raw ${CMAKE_CURRENT_BINARY_DIR}/boot-link)
    set(final ${CMAKE_BINARY_DIR}/boot-web)
    set(helper ${root}/tools/browser/tail-call-postlink.mjs)
    set(postlink ${CMAKE_CURRENT_BINARY_DIR}/wasm-postlink.json)
    set(manifest ${CMAKE_CURRENT_BINARY_DIR}/boot-manifest.mjs)
    set(stamp ${CMAKE_CURRENT_BINARY_DIR}/boot-publication.complete)
    set(state ${CMAKE_CURRENT_BINARY_DIR}/boot-publication.policy)
    file(SHA256 ${postlink} postlink_identity)
    file(SHA256 ${manifest} manifest_identity)
    set(policy "${postlink_identity};${manifest_identity}")
    set(previous_policy "")
    if(EXISTS ${state})
        file(READ ${state} previous_policy)
    endif()
    if(NOT previous_policy STREQUAL policy)
        file(REMOVE ${stamp})
        file(WRITE ${state} "${policy}")
    endif()
    set_target_properties(story_boot_link PROPERTIES OUTPUT_NAME story_boot RUNTIME_OUTPUT_DIRECTORY ${raw} SUFFIX .mjs)
    add_custom_command(TARGET story_boot_link POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E true
        BYPRODUCTS ${raw}/story_boot.wasm
        VERBATIM)
    set(dependencies story_boot_link ${raw}/story_boot.wasm ${helper} ${postlink} ${manifest})
    if(BROWSER_ACTUAL_TAIL_CALLS)
        list(APPEND dependencies "${BROWSER_WASM_OPT}")
    endif()
    add_custom_command(OUTPUT ${stamp} ${final}/story_boot.wasm ${final}/story_boot.mjs ${final}/boot-manifest.mjs
        COMMAND ${NODE_EXECUTABLE} ${helper} ${raw}/story_boot.wasm ${final}/story_boot.wasm ${postlink}
            ${raw}/story_boot.mjs ${final}/story_boot.mjs ${manifest} ${final}/boot-manifest.mjs
        COMMAND ${CMAKE_COMMAND} -E touch ${stamp}
        DEPENDS ${dependencies}
        VERBATIM)
    add_custom_target(story_boot ALL DEPENDS ${stamp} ${final}/story_boot.wasm ${final}/story_boot.mjs ${final}/boot-manifest.mjs)
endfunction()
